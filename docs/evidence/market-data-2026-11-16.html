Reproducible read-only query

SELECT o.id AS observation_id, o.provider, o.ticker, o.source_time_utc,
json_extract(o.payload_json, '$.price') AS price,
json_extract(o.payload_json, '$.day_volume') AS cumulative_day_volume,
(SELECT min(r.id) FROM receipts r WHERE r.observation_id=o.id) AS first_receipt_id,
(SELECT r.received_at_utc FROM receipts r WHERE r.observation_id=o.id ORDER BY r.id LIMIT 1) AS first_received_at_utc
FROM observations o ORDER BY o.id DESC LIMIT 25
